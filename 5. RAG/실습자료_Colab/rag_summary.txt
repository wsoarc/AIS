from bs4 import BeautifulSoup
from blingfire import text_to_sentences_and_offsets

def parse_htmls(search_results):
    documents = []
    for result in search_results:
        soup = BeautifulSoup(result["page_result"], features="lxml")
        documents.append(soup.get_text(" ", strip=True))
    return documents

7. LlamaIndex Retriever
from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import Document
from llama_index.embeddings.openai import OpenAIEmbedding

Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

class LlamaIndexRetriever:
    def __init__(self):
        self.parser = SentenceSplitter(chunk_size=512, chunk_overlap=0)

    def retrieve(self, query, search_results, topk):
        documents = [Document(text=text) for text in parse_htmls(search_results)]
        index = VectorStoreIndex.from_documents(
            documents=documents,
            transformations=[self.parser]
        )
        retriever = index.as_retriever(similarity_top_k=topk)
        nodes = retriever.retrieve(query)
        return [node.node.get_content().strip() for node in nodes]

8. Reader: 검색 문맥을 Prompt로 구성하고 답변 생성
class Reader:
    def prompt_generator(self, query, top_k_chunks):
        references = "# References\n"
        for chunk in top_k_chunks:
            references += f"- {chunk.strip()}\n"
        references = references[:MAX_CONTEXT_REFERENCES_LENGTH]

        return [
            {"role": "system", "content": self.system_prompt},
            {"role": "user", "content":
             f"{references}\n------\n"
             f"Using only the references listed above, answer the question.\n"
             f"Question: {query}"}
        ]

    def generate_response(self, query, top_k_chunks):
        return oai_client.chat.completions.create(
            model="gpt-3.5-turbo",
            temperature=0,
            messages=self.prompt_generator(query, top_k_chunks)
        ).choices[0].message.content

9. Retriever와 Reader를 결합한 RAG
class RAG:
    def __init__(self):
        self.retriever = LlamaIndexRetriever()
        self.reader = Reader()

    def inference(self, query, search_results, topk):
        retrieved_results = self.retriever.retrieve(query, search_results, topk)
        answer = self.reader.generate_response(query, retrieved_results)
        return answer, retrieved_results

12. MCP Tool을 사용하는 Agent
class KGQueryEngineWithMCP:
    def __init__(self, mcp_tool_spec, model, llm=None):
        self.llm = llm or OpenAI(model=model, temperature=0)
        self.mcp_tool_spec = mcp_tool_spec
        self.agent = None
        self.agent_context = None

    async def init_agent(self):
        tools = await self.mcp_tool_spec.to_tool_list_async()
        self.agent = FunctionAgent(
            name="Agent",
            description="An agent that can work with Our Knowledge Graph api.",
            tools=tools,
            llm=self.llm,
            system_prompt=SYSTEM_PROMPT
        )
        self.agent_context = Context(self.agent)

    async def query(self, question):
        handler = self.agent.run(question, ctx=self.agent_context)
        response = await handler
        return str(response)

13. MCP 기반 RAG
class RAGwithMCP:
    def __init__(self, mcp_tool):
        self.mcp_application = KGQueryEngineWithMCP(mcp_tool, model="gpt-4o-mini")
        self.reader = Reader()

    async def retrieve(self, query, query_time, search_results, topk):
        await self.mcp_application.init_agent()
        full_query = f"Query: {query}\nQuery time: {query_time}"
        return await self.mcp_application.query(full_query)

    async def inference(self, query, search_results, query_time, topk):
        retrieved_results = await self.retrieve(query, query_time, search_results, topk)
        references = retrieved_results if isinstance(retrieved_results, list) else [retrieved_results]
        answer = self.reader.generate_response(query, query_time, references)
        return {"retrieved_results": retrieved_results, "answer": answer}
